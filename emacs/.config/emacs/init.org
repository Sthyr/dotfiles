* Fundamental
** Font
The font and style of the font are determinated by the face, each comment, code, header, latex code, etc uses a different font face.
`default': The face uses by default
`fixed-pitch': monospace font. Each character uses the same space
`variable-pitch': The characters have different spaces, different width.
Emacs always uses the default face unless we explicity decided which font to use on each case(a face for comments, another for code, etc.)
#+BEGIN_SRC emacs-lisp
      (set-face-attribute 'default nil
      		    :family "Iosevka"
      		    ;; Height = point size * 10
      		    :height 180
      		    :weight 'Regular)
      (set-face-attribute 'fixed-pitch nil
    		      :family "Iosevka"
  		    :height 180)
      (set-face-attribute 'variable-pitch nil
      		    :family "Iosevka Etoile"
      		    :height 180)
#+END_SRC

(use-package tempel
  :ensure t
  ;; By default, tempel looks at the file "templates" in
  ;; user-emacs-directory, but you can customize that with the
  ;; tempel-path variable:z
  ;; :custom
  ;; (tempel-path (concat user-emacs-directory "custom_template_file"))
  :bind (("M-´" . tempel-insert)
         ("M-+" . tempel-complete)
         :map tempel-map
         ("C-c RET" . tempel-done)
         ("C-<down>" . tempel-next)
         ("C-<up>" . tempel-previous)
         ("M-<down>" . tempel-next)
         ("M-<up>" . tempel-previous))
  :init
  ;; Make a function that adds the tempel expansion function to the
  ;; list of completion-at-point-functions (capf).
  (defun tempel-setup-capf ()
    (add-hook 'completion-at-point-functions #'tempel-expand -1 'local))
  ;; Put tempel-expand on the list whenever you start programming or
  ;; writing prose.
  (add-hook 'prog-mode-hook 'tempel-setup-capf)
  (add-hook 'text-mode-hook 'tempel-setup-capf))

** AUCTeX
#+BEGIN_SRC emacs-lisp
(use-package tex
  :ensure auctex
  :hook ((LaTeX-mode . turn-on-reftex)            ; referencias cruzadas
         (LaTeX-mode . TeX-source-correlate-mode) ; sincroniza código <-> PDF (SyncTeX)
         (LaTeX-mode . visual-line-mode)          ; ajusta líneas largas
         (LaTeX-mode . (lambda () (variable-pitch-mode -1)))) ; código .tex en monoespaciada
  :custom
  (TeX-auto-save t)
  (TeX-parse-self t)
  (TeX-source-correlate-start-server t)
  (reftex-plug-into-AUCTeX t)
  ;; Visor por defecto; se alterna con C-c v
  (TeX-view-program-selection '((output-pdf "PDF Tools")))
  :config
  ;; Recargar el PDF en pdf-tools después de compilar
  (add-hook 'TeX-after-compilation-finished-functions
            #'TeX-revert-document-buffer)

  (defun sthyr/TeX-alternar-visor ()
    "Alternar el visor PDF entre PDF Tools y Okular."
    (interactive)
    (let ((nuevo (if (equal (cadr (assq 'output-pdf TeX-view-program-selection))
                            "PDF Tools")
                     "Okular"
                   "PDF Tools")))
      (setq TeX-view-program-selection `((output-pdf ,nuevo)))
      (message "Visor PDF: %s" nuevo)))

  (with-eval-after-load 'latex
    (keymap-set LaTeX-mode-map "C-c v" #'sthyr/TeX-alternar-visor)))
#+END_SRC
